⚡ HIZLI KOMUTLAR:

# Kurulum
./setup.sh

# API test
python3 test-config.py

# Telegram Bot'u başlat (arka planda)
./start-bot.sh

# Tarama (menü)
./quick-start.sh

# Tarama (direkt - zafiyet bulana kadar!)
source venv/bin/activate
python3 scanner.py

# Bot'a Telegram'dan komut gönder:
/start   - Bot'u başlat
/stats   - İstatistikler
/cvss    - CVSS skoru sorgula
/latest  - Son zafiyet
/help    - Komutlar

# Bot'u durdur
./stop-bot.sh

# İstatistikler
cat scanned_plugins.json

# Log izle
tail -f logs/scanner.log
tail -f logs/telegram_bot.log
